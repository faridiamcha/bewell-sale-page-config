    async function buildCard(def, type) {
      const container=$(type==='addon'?'#bw-addon-list':'#bw-related-list');
      const card=node('article','bw-product-card'),img=node('img'),title=node('h3','',def.name),price=node('div','bw-card-price','กำลังอ่านราคา…'),fields=node('div','bw-card-fields');
      card.append(img,title,price,fields);img.loading='lazy';img.width=400;img.height=300;
      const link=node('a','bw-small','ดูรายละเอียดสินค้า');link.href=sameStore(def.path).href;card.append(link);
      let checkbox,button;
      if(type==='addon') { const label=node('label','bw-check');checkbox=node('input');checkbox.type='checkbox';checkbox.disabled=true;label.append(checkbox,document.createTextNode('เลือกซื้อคู่ 1 ชิ้น'));card.append(label); }
      else { button=node('button','bw-outline','เพิ่มลงตะกร้า');button.type='button';button.disabled=true;card.append(button); }
      container.append(card);
      try {
        const p=await product(def);const controls=[];
        p.attributes.forEach(a=>{
          const label=node('label','',a.label),select=node('select');select.setAttribute('aria-label',def.name+' '+a.label);select.dataset.attribute=a.id;
          const empty=node('option','','กรุณาเลือก '+a.label);empty.value='';select.append(empty);
          a.options.forEach(o=>{const option=node('option','',o.label);option.value=String(o.id);select.append(option);});label.append(select);fields.append(label);controls.push(select);
        });
        const getSelection=()=>choice(p,Object.fromEntries(controls.map(select=>[select.dataset.attribute,select.value])));
        function updateCard() {
          // Disable ชุด Attribute ที่ไม่มี child ตรงกัน โดยยังคงเลือกค่าอื่นใหม่ได้
          controls.forEach(select=>Array.from(select.options).forEach(option=>{
            if(!option.value)return;
            const partial=Object.fromEntries(controls.filter(other=>other!==select&&other.value).map(other=>[other.dataset.attribute,other.value]));partial[select.dataset.attribute]=option.value;
            option.disabled=!Object.keys(p.config.index).some(child=>Object.entries(partial).every(([a,v])=>String(p.config.index[child][a])===String(v)));
          }));
          const selected=getSelection();price.textContent=selected?money(selected.price):(p.outOfStock?'สินค้าหมด':'เลือกตัวเลือกเพื่อดูราคา');
          setImage(img,selected?.image||p.defaultImage,def.name);
          if(checkbox){checkbox.disabled=state.busy||state.locked||!selected;checkbox.dataset.unavailable=selected?'0':'1';if(!selected)checkbox.checked=false;}
          refresh();
        }
        controls.forEach(select=>select.addEventListener('change',updateCard));
        if(type==='addon') {
          const item={...p,checkbox,getSelection};state.addons.push(item);checkbox.addEventListener('change',refresh);
          if(def.dealText)card.insertBefore(node('p','bw-deal-tag',def.dealText),fields);
        } else { const item={...p,button,getSelection};state.related.push(item);button.addEventListener('click',()=>buy([{product:p,selection:getSelection(),qty:1}])); }
        updateCard();
      } catch(error) {price.textContent='ยังไม่พร้อมจำหน่ายจากหน้านี้';card.append(node('p','bw-small','กรุณาดูรายละเอียดที่หน้าสินค้า หรือแจ้งผู้ดูแลให้ตรวจ URL / Product ID'));img.hidden=true;}
    }
    function formKey() { const match=document.cookie.match(/(?:^|;\s*)form_key=([^;]+)/);try{return match?decodeURIComponent(match[1]):'';}catch{return '';} }
    async function cart() {
      const url=endpoint('customer/section/load/');url.searchParams.set('sections','cart');url.searchParams.set('force_new_section_timestamp','true');url.searchParams.set('_',String(Date.now()));
      const res=await request(url.href,{cache:'no-store',headers:{'X-Requested-With':'XMLHttpRequest','Accept':'application/json'}});
      const data=await res.json();if(!data.cart || !Array.isArray(data.cart.items))throw new Error('ไม่สามารถตรวจสถานะตะกร้าได้');return data.cart;
    }
    function cartQty(snapshot, item) {
      return snapshot.items.filter(line=>{
        const id=String(line.product_id);if(id!==String(item.product.def.id) && id!==String(item.selection.child))return false;
        return item.selection.labels.every(expect=>(line.options||[]).some(o=>normalizeHTML(o.label)===norm(expect.label)&&normalizeHTML(o.value)===norm(expect.value)));
      }).reduce((total,line)=>total+(Number(line.qty)||0),0);
    }
    async function addOne(item) {
      const token=formKey();if(!token)throw new Error('ยังไม่มีเซสชันตะกร้า กรุณารีโหลดหน้า');
      const before=await cart();
      const body=new URLSearchParams({form_key:token,product:String(item.product.def.id),qty:String(item.qty)});
      Object.entries(item.selection.attributes).forEach(([a,v])=>body.set('super_attribute['+a+']',v));
      const res=await request(endpoint('checkout/cart/add/').href,{method:'POST',cache:'no-store',headers:{'Content-Type':'application/x-www-form-urlencoded; charset=UTF-8','X-Requested-With':'XMLHttpRequest','Accept':'application/json'},body:body.toString()});
      // HTTP 200 อย่างเดียวไม่ได้ยืนยันว่าเพิ่มสำเร็จ ต้องอ่าน JSON และตรวจ qty ที่ตะกร้า
      let result;try{result=await res.json();}catch{throw new Error('ผลตอบกลับไม่ใช่ JSON กรุณาตรวจตะกร้า');}
      if(result?.error || result?.errors)throw new Error('Magento ไม่ยืนยันรายการที่เลือก กรุณาตรวจสต็อกและตัวเลือกในตะกร้า');
      // ร้านที่เปิด Redirect to Cart อาจคืน backUrl แม้สำเร็จ จึงไม่ใช้ URL เป็นหลักฐานสำเร็จ/ล้มเหลว
      // ไม่ตาม redirect นี้เอง ตรวจว่าอยู่ร้านเดียวกัน แล้วใช้จำนวนจริงใน cart section เป็นหลัก
      if(result?.backUrl)sameStore(result.backUrl);
      const after=await cart();
      if(cartQty(after,item)-cartQty(before,item)!==item.qty)throw new Error('ยังยืนยันจำนวนสินค้าที่เพิ่มไม่ได้ กรุณาตรวจตะกร้าก่อนลองใหม่');
    }
    async function refreshMiniCart() {
      if(typeof window.require!=='function')return;
      // ใช้ Magento customer-data invalidate/reload เพื่อให้ badge/minicart อ่านจากเซิร์ฟเวอร์
      window.require(['Magento_Customer/js/customer-data'],function(customerData){customerData.invalidate(['cart']);customerData.reload(['cart'],true);});
    }
    async function buy(items) {
      if(state.busy||state.locked||!CONFIG.cartEnabled)return;
      if(location.origin!==CONFIG.allowedOrigin){message('หน้าและตะกร้าต้องอยู่โดเมนร้านที่กำหนด',true);return;}
      if(!items.length || items.some(i=>!i.selection||!Number.isInteger(i.qty)||i.qty<1||i.qty>CONFIG.maxQty)){message('กรุณาเลือกตัวเลือกและจำนวนให้ครบ',true);return;}
      state.busy=true;refresh();root.setAttribute('aria-busy','true');
      try {
        if(!formKey())throw new Error('ยังไม่มีเซสชันตะกร้า กรุณารีโหลดหน้า');
        for(let index=0;index<items.length;index++){message('กำลังเพิ่มสินค้า '+(index+1)+' / '+items.length+' …');await addOne(items[index]);}
        refreshMiniCart();message('เพิ่มสินค้าที่เลือกครบแล้ว กรุณาตรวจยอดสุทธิและสิทธิ์โปรโมชั่นในตะกร้า');
        state.locked=true; // กันการกดซ้ำในหน้าที่เพิ่มสำเร็จแล้ว
        if(CONFIG.redirectToCart)location.assign(endpoint('checkout/cart/').href);else showReload();
      } catch(error) {
        // ไม่มี automatic retry/rollback เพราะรายการก่อนหน้าอาจสำเร็จแล้ว
        state.locked=true;message((error.name==='AbortError'?'การเชื่อมต่อใช้เวลานานเกินกำหนด':error.message)+' — '+TEXT.partial,true);showReload();refreshMiniCart();
      } finally {state.busy=false;root.removeAttribute('aria-busy');refresh();}
    }
